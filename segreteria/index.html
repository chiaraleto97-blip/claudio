Firme dal telefono

Non collegato

Proiettate il QR: ogni docente lo inquadra, scrive nome e cognome sul proprio telefono e firma con il dito. Le firme arrivano qui da sole, ogni pochi secondi, e compaiono nel foglio da stampare. Chi non è ancora nell'elenco viene aggiunto.

Collegamento al registro (si fa una volta)
  1. Aprite script.google.com, Nuovo progetto, incollate lo script del registro (pulsante qui sotto) e salvate.
  2. Scegliete la funzione prepara e premete Esegui: nel registro in basso compaiono il link del Foglio e la chiave.
  3. Esegui il deployment, Nuovo deployment, tipo App web: esegui come Me, accesso Chiunque. Copiate l'URL dell'app web.
  4. Incollate qui l'URL e la chiave.
/**
 * IDEA^IA · Registro del corso
 * Riceve le firme dal telefono dei corsisti e le conserva in un Foglio Google
 * di chi conduce il corso.
 *
 * Come si prepara (una volta sola):
 * 1. script.google.com > Nuovo progetto > incollate tutto questo testo > Salva.
 * 2. Scegliete la funzione «prepara» e premete Esegui (autorizzate con il vostro account).
 *    Nel Registro di esecuzione trovate il link del Foglio e la CHIAVE.
 * 3. Esegui il deployment > Nuovo deployment > tipo «App web».
 *    Esegui come: Me. Chi ha accesso: Chiunque. Poi Esegui il deployment e copiate l'URL dell'app web.
 * 4. Segreteria del corso, scheda Foglio firme: incollate l'URL e la CHIAVE.
 *
 * Le firme si leggono solo con la CHIAVE.
 */
const P = PropertiesService.getScriptProperties();

function prepara() {
  let ss, id = P.getProperty('FOGLIO');
  if (id) ss = SpreadsheetApp.openById(id);
  else { ss = SpreadsheetApp.create('IDEAᴵᴬ · Registro del corso'); P.setProperty('FOGLIO', ss.getId()); }
  foglio_(ss, 'Firme', ['Quando', 'Codice corso', 'Incontro', 'Momento', 'Nome e cognome', 'Firma (immagine)']);
  const vuoto = ss.getSheetByName('Foglio1') || ss.getSheetByName('Sheet1');
  if (vuoto && ss.getSheets().length > 1) ss.deleteSheet(vuoto);
  let k = P.getProperty('CHIAVE');
  if (!k) { k = Utilities.getUuid().replace(/-/g, '').slice(0, 12); P.setProperty('CHIAVE', k); }
  Logger.log('Foglio del registro: ' + ss.getUrl());
  Logger.log('CHIAVE da incollare nella Segreteria: ' + k);
}

function foglio_(ss, nome, intestazione) {
  let sh = ss.getSheetByName(nome);
  if (!sh) { sh = ss.insertSheet(nome); sh.appendRow(intestazione); sh.setFrozenRows(1); }
  return sh;
}
const ss_ = () => SpreadsheetApp.openById(P.getProperty('FOGLIO'));
const out_ = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
const codice_ = c => /^[a-z0-9]{4,12}$/.test(c || '') ? c : null;
/* testo pulito, e mai interpretato come formula dal Foglio */
const pulito_ = (t, n) => {
  const s = String(t == null ? '' : t).replace(/[\u0000-\u001f]/g, ' ').trim().slice(0, n);
  return /^[=+\-@]/.test(s) ? "'" + s : s;
};
const ms_ = d => d instanceof Date ? d.getTime() : Number(d) || 0;

function doGet(e) {
  const p = e.parameter || {};
  try {
    if (p.azione === 'firme') {
      if (!p.chiave || p.chiave !== P.getProperty('CHIAVE')) return out_({ errore: 'chiave' });
      const c = codice_(p.codice); if (!c) return out_({ errore: 'codice' });
      const righe = ss_().getSheetByName('Firme').getDataRange().getValues().slice(1).filter(r => r[1] === c);
      return out_({ firme: righe.map(r => ({ quando: ms_(r[0]), incontro: Number(r[2]), momento: r[3] === 'uscita' ? 'u' : 'e', nome: String(r[4]).replace(/^'/, ''), firma: r[5] })) });
    }
    return out_({ ok: true, servizio: 'IDEA^IA registro' });
  } catch (err) { return out_({ errore: String(err) }); }
}

function doPost(e) {
  let d;
  try { d = JSON.parse(e.postData.contents); } catch (err) { return out_({ errore: 'formato' }); }
  const lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    if (d.tipo === 'firma') {
      const c = codice_(d.codice), nome = pulito_(d.nome, 80), inc = Number(d.incontro), f = String(d.firma || '');
      if (!c || nome.length < 3 || !(inc >= 1 && inc <= 5) || !/^data:image\/png;base64,[A-Za-z0-9+\/=]+$/.test(f) || f.length > 45000)
        return out_({ errore: 'dati' });
      const sh = ss_().getSheetByName('Firme');
      if (sh.getLastRow() > 3000) return out_({ errore: 'pieno' });
      sh.appendRow([new Date(), c, inc, d.momento === 'u' ? 'uscita' : 'entrata', nome, f]);
      return out_({ ok: true });
    }
    return out_({ errore: 'tipo' });
  } catch (err) { return out_({ errore: String(err) }); }
  finally { lock.releaseLock(); }
}

Firma qui, sullo schermo

In alternativa, sullo schermo touch della LIM o su un tablet: ognuno tocca il proprio nome e firma con il dito. La firma compare nel foglio qui sotto e segna anche la presenza.

Firmate con il dito, o con la penna, sopra la riga.

La fascia di cielo in alto si stampa solo con «Grafica di sfondo» attiva nella finestra di stampa.

Online, con Google Moduli

Lo script crea da solo il modulo anonimo, con le stesse domande del foglio qui sotto, e il foglio Google delle risposte. Si fa una volta sola.

  1. Aprite script.google.com con l'account della scuola e scegliete Nuovo progetto.
  2. Cancellate il testo che c'è e incollate lo script (pulsante qui sotto).
  3. Salvate, poi premete Esegui e autorizzate.
  4. Nel registro in basso compaiono tre link: quello «da dare ai corsisti» va incollato nella slide Questionario finale della presentazione, che lo trasforma in un QR.
/**
 * IDEA^IA · Questionario finale
 * Crea il modulo Google anonimo del questionario e il foglio delle risposte.
 * Come si usa: script.google.com > Nuovo progetto > incollate tutto questo testo >
 * Salva > Esegui (funzione creaQuestionario) > autorizzate con il vostro account.
 * Nel Registro di esecuzione trovate i tre link: quello "da dare ai corsisti"
 * va incollato nella slide "Questionario finale" della presentazione.
 */
function creaQuestionario() {
  const scala = ['1', '2', '3', '4', '5'];
  const f = FormApp.create('IDEAᴵᴬ · Questionario finale');
  f.setDescription('Il questionario è anonimo e richiede circa tre minuti. ' +
    'Nelle griglie: 1 = per niente, 5 = moltissimo. Grazie!');
  f.setCollectEmail(false);
  f.setProgressBar(true);
  f.setConfirmationMessage('Grazie! Le vostre risposte aiutano a migliorare il corso.');

  f.addSectionHeaderItem().setTitle('Il corso');
  f.addGridItem()
    .setTitle('Quanto siete d\'accordo con queste frasi?')
    .setRows([
      'Gli obiettivi del corso erano chiari',
      'I contenuti sono utili per la mia didattica',
      'La teoria sui modelli linguistici mi ha aiutato a capire come funziona Claude',
      'Il laboratorio (carte, scheda progetto, costruzione) è stato efficace',
      'I materiali (Claudio, carte, schede) sono stati utili',
      'Tempi e organizzazione degli incontri erano adeguati',
      'Il clima di lavoro favoriva la collaborazione',
      'La conduzione è stata chiara e disponibile'
    ])
    .setColumns(scala)
    .setRequired(true);

  f.addSectionHeaderItem().setTitle('Prima e dopo');
  const abilita = [
    'Costruire uno strumento didattico con l\'IA',
    'Usare l\'IA in modo responsabile (privacy, verifiche)'
  ];
  f.addGridItem().setTitle('PRIMA del corso, quanto vi sentivate capaci di…')
    .setRows(abilita).setColumns(scala).setRequired(true);
  f.addGridItem().setTitle('ADESSO, quanto vi sentite capaci di…')
    .setRows(abilita).setColumns(scala).setRequired(true);

  f.addSectionHeaderItem().setTitle('In classe');
  f.addMultipleChoiceItem()
    .setTitle('Avete provato il vostro strumento in classe?')
    .setChoiceValues(['Sì, più volte', 'Sì, una volta', 'Non ancora, ma lo farò', 'No'])
    .setRequired(true);
  f.addScaleItem()
    .setTitle('Quanto è probabile che userete ancora l\'IA per creare strumenti per la classe?')
    .setBounds(1, 5).setLabels('Per niente', 'Moltissimo')
    .setRequired(true);

  f.addSectionHeaderItem().setTitle('In parole vostre');
  f.addParagraphTextItem().setTitle('Che cosa vi è stato più utile?');
  f.addParagraphTextItem().setTitle('Che cosa cambiereste?');
  f.addParagraphTextItem().setTitle('Su quali temi vorreste un secondo livello del corso?');
  f.addMultipleChoiceItem()
    .setTitle('Ordine di scuola (facoltativo)')
    .setChoiceValues(['Infanzia', 'Primaria', 'Secondaria di primo grado', 'Altro']);

  const ss = SpreadsheetApp.create('IDEAᴵᴬ · Risposte al questionario finale');
  f.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());

  Logger.log('Link da dare ai corsisti: ' + f.getPublishedUrl());
  Logger.log('Per modificare il modulo: ' + f.getEditUrl());
  Logger.log('Foglio delle risposte: ' + ss.getUrl());
}
